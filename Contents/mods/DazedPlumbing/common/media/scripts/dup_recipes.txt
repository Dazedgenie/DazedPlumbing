    /* ------------------------------------------------------------------ *
     * Dazed Utilities: Plumbing -- the recipes.
     *
     * CRAFTED tanks are properly welded: sheet metal, pipe, a torch and a
     * mask, Welding skill by size. SALVAGED ones are rigged from small
     * sheets, screws and tape: no welding gear, a little skill, and the
     * result is the worn, leakier kind. Everything here uses item types the
     * vanilla metalworking recipes use.
     * Not learn-gated: any survivor with the skill and the parts can build one.
     * ------------------------------------------------------------------ */

    craftRecipe MakeDazedTankSmallPropaneSalvaged
    {
        timedAction     = Welding,
        time            = 500,
        SkillRequired   = MetalWelding:1,
        Tags            = AnySurfaceCraft,
        category        = Welding,
        xpAward         = MetalWelding:11,
        inputs
        {
            item 1 tags[base:screwdriver] mode:keep flags[Prop1],
            item 6 [Base.SmallSheetMetal],
            item 6 [Base.Screws],
            item 2 [Base.DuctTape],
        }
        outputs
        {
            item 1 Base.DazedTankSmallPropaneSalvaged,
        }
    }
